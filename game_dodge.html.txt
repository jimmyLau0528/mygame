    <div id="scoreBoard">得分: <span id="score">0</span></div>
    <canvas id="gameCanvas" width="360" height="500"></canvas>
    
    <div id="gameOverText">
        GAME OVER<br>
        <span style="font-size: 18px; color: white;">點擊畫面任何地方重新開始</span>
    </div>

    <div class="hint">🕹️ 玩法：長按【畫面左邊】向左移，長按【畫面右邊】向右移！</div>

    <script>
        const canvas = document.getElementById("gameCanvas");
        const ctx = canvas.getContext("2d");
        const scoreDisplay = document.getElementById("score");
        const gameOverText = document.getElementById("gameOverText");

        let score = 0;
        let gameOver = false;

        const player = {
            x: 155,
            y: 430,
            width: 50,
            height: 50,
            speed: 7,
            color: "#00adb5"
        };

        const obstacles = [];
        const obstacleWidth = 45;
        const obstacleHeight = 45;
        let spawnTimer = 0;
        let spawnInterval = 35; 

        let moveLeft = false;
        let moveRight = false;

        window.addEventListener("touchstart", (e) => {
            if (gameOver) {
                resetGame();
                return;
            }
            const touchX = e.touches[0].clientX;
            const screenWidth = window.innerWidth;
            
            if (touchX < screenWidth / 2) {
                moveLeft = true;
                moveRight = false;
            } else {
                moveRight = true;
                moveLeft = false;
            }
        });

        window.addEventListener("touchend", () => {
            moveLeft = false;
            moveRight = false;
        });

        function resetGame() {
            score = 0;
            gameOver = false;
            player.x = 155;
            obstacles.length = 0;
            spawnTimer = 0;
            scoreDisplay.innerText = score;
            gameOverText.style.display = "none";
            update();
        }

        function checkCollision(rect1, rect2) {
            return rect1.x < rect2.x + rect2.width &&
                   rect1.x + rect1.width > rect2.x &&
                   rect1.y < rect2.y + rect2.height &&
                   rect1.y + rect1.height > rect2.y;
        }

        function update() {
            if (gameOver) return;

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            if (moveLeft && player.x > 0) {
                player.x -= player.speed;
            }
            if (moveRight && player.x < canvas.width - player.width) {
                player.x += player.speed;
            }

            ctx.fillStyle = player.color;
            ctx.fillRect(player.x, player.y, player.width, player.height);

            spawnTimer++;
            if (spawnTimer >= spawnInterval) {
                spawnTimer = 0;
                const randomX = Math.random() * (canvas.width - obstacleWidth);
                obstacles.push({ x: randomX, y: -obstacleHeight, width: obstacleWidth, height: obstacleHeight });
            }

            for (let i = obstacles.length - 1; i >= 0; i--) {
                obstacles[i].y += (5 + score * 0.25); 

                ctx.fillStyle = "#ff2e63";
                ctx.fillRect(obstacles[i].x, obstacles[i].y, obstacles[i].width, obstacles[i].height);

                if (checkCollision(player, obstacles[i])) {
                    gameOver = true;
                    gameOverText.style.display = "block";
                }

                if (obstacles[i].y > canvas.height) {
                    obstacles.splice(i, 1);
                    score++;
                    scoreDisplay.innerText = score;
                }
            }

            requestAnimationFrame(update);
        }

        update();
    </script>
